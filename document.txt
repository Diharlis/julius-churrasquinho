        // Canal de comunicação em tempo real entre abas no mesmo navegador
        const orderChannel = new BroadcastChannel('julius_orders_channel');

        // Escutar eventos de sincronização em tempo real entre telas/abas
        orderChannel.onmessage = (event) => {
            if (!event.data) return;
            const { type, pedido, id, novoStatus } = event.data;

            switch (type) {
                case 'NOVO_PEDIDO':
                    adicionarPedidoAoGestor(pedido, true, false);
                    break;
                case 'STATUS_ALTERADO':
                    alterarStatusPedido(id, novoStatus, false);
                    break;
                case 'PEDIDO_REMOVIDO':
                    removerPedidoGestor(id, false);
                    break;
                case 'CONCLUIDOS_LIMPOS':
                    limparConcluidos(false);
                    break;
            }
        };

        // Sistema Visual de Notificações Toast
        function showToast(message, type = 'info') {
            let container = document.getElementById('toast-container');
            if (!container) {
                container = document.createElement('div');
                container.id = 'toast-container';
                container.className = 'fixed top-20 right-4 z-50 flex flex-col gap-2 pointer-events-none';
                document.body.appendChild(container);
            }

            const toast = document.createElement('div');
            const bgClass = type === 'success' ? 'bg-emerald-600' : type === 'error' ? 'bg-brand-red' : 'bg-slate-800 text-brand-gold border-brand-gold/40';
            toast.className = `${bgClass} text-white text-xs md:text-sm font-semibold px-4 py-2.5 rounded-xl shadow-2xl border border-slate-700 transition-all duration-300 opacity-0 transform translate-y-2 pointer-events-auto flex items-center gap-2`;
            toast.innerHTML = `<i class="${type === 'success' ? 'fa-solid fa-circle-check' : type === 'error' ? 'fa-solid fa-circle-exclamation' : 'fa-solid fa-circle-info'}"></i> <span>${message}</span>`;

            container.appendChild(toast);

            setTimeout(() => {
                toast.classList.remove('opacity-0', 'translate-y-2');
            }, 10);

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 2600);
        }

        // --- LÓGICA DO CARRINHO / COMANDA ---

        function addToCart(id, name, price) {
            const existingItem = cart.find(item => item.id === id);
            if (existingItem) {
                existingItem.qty += 1;
            } else {
                cart.push({ id, name, price, qty: 1 });
            }
            updateCartUI();
            showToast(`"${name}" adicionado à comanda!`, 'success');
        }

        function updateQty(id, change) {
            const item = cart.find(item => item.id === id);
            if (item) {
                item.qty += change;
                if (item.qty <= 0) {
                    cart = cart.filter(i => i.id !== id);
                }
            }
            updateCartUI();
        }

        function removeItem(id) {
            const item = cart.find(i => i.id === id);
            cart = cart.filter(item => item.id !== id);
            updateCartUI();
            if (item) showToast(`"${item.name}" removido da comanda.`, 'info');
        }

        function clearCart() {
            if (cart.length === 0) return;
            cart = [];
            updateCartUI();
            showToast("Comanda esvaziada.", 'info');
        }

        function calculateTotal() {
            return cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
        }

        function calculateTotalItems() {
            return cart.reduce((sum, item) => sum + item.qty, 0);
        }

        function updateCartUI() {
            const total = calculateTotal();
            const totalItems = calculateTotalItems();

            const headerBadge = document.getElementById('header-cart-badge');
            if (totalItems > 0) {
                headerBadge.innerText = totalItems;
                headerBadge.classList.remove('hidden');
            } else {
                headerBadge.classList.add('hidden');
            }

            const countElem = document.getElementById('selected-count');
            const totalElem = document.getElementById('order-total');

            if (totalItems === 0) {
                countElem.innerText = "Comanda Vazia";
                totalElem.innerText = "R$ 0,00";
            } else {
                countElem.innerText = `${totalItems} item(ns) selecionado(s)`;
                totalElem.innerText = `R$ ${total.toFixed(2).replace('.', ',')}`;
            }

            renderCartModalList();
        }

        function renderCartModalList() {
            const container = document.getElementById('cart-items-container');
            const modalTotal = document.getElementById('modal-cart-total');
            const total = calculateTotal();

            modalTotal.innerText = `R$ ${total.toFixed(2).replace('.', ',')}`;

            if (cart.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-8 text-slate-500 space-y-2">
                        <i class="fa-solid fa-basket-shopping text-4xl"></i>
                        <p class="text-xs md:text-sm">Sua comanda está vazia.<br>Escolha opções saborosas do cardápio!</p>
                    </div>
                `;
                return;
            }

            let html = '';
            cart.forEach(item => {
                const subtotal = item.price * item.qty;
                html += `
                    <div class="flex items-center justify-between bg-slate-900 p-3 rounded-2xl border border-brand-border text-xs md:text-sm">
                        <div class="flex-1 pr-2">
                            <h4 class="font-bold text-slate-100">${item.name}</h4>
                            <span class="text-brand-gold font-semibold">R$ ${item.price.toFixed(2).replace('.', ',')} un</span>
                        </div>

                        <div class="flex items-center gap-2 bg-brand-card px-2.5 py-1 rounded-xl border border-slate-800">
                            <button onclick="updateQty(${item.id}, -1)" class="w-6 h-6 bg-slate-800 hover:bg-brand-red text-white rounded-lg font-bold flex items-center justify-center transition">-</button>
                            <span class="font-bold text-slate-200 w-4 text-center">${item.qty}</span>
                            <button onclick="updateQty(${item.id}, 1)" class="w-6 h-6 bg-slate-800 hover:bg-brand-red text-white rounded-lg font-bold flex items-center justify-center transition">+</button>
                        </div>

                        <div class="text-right pl-3">
                            <span class="font-bold text-slate-100 block">R$ ${subtotal.toFixed(2).replace('.', ',')}</span>
                            <button onclick="removeItem(${item.id})" class="text-slate-500 hover:text-brand-red text-xs mt-0.5">
                                <i class="fa-solid fa-trash-can"></i>
                            </button>
                        </div>
                    </div>
                `;
            });

            container.innerHTML = html;
        }

        function toggleCartModal(show) {
            const modal = document.getElementById('cart-modal');
            if (show) {
                renderCartModalList();
                modal.classList.remove('hidden');
            } else {
                modal.classList.add('hidden');
            }
        }

        // ENVIAR PEDIDO: WhatsApp + Transmissão ao vivo para a Cozinha
        function sendWhatsAppOrder() {
            if (cart.length === 0) {
                showToast("Sua comanda está vazia!", 'error');
                return;
            }

            const clientName = document.getElementById('cart-client-name').value.trim() || "Cliente Balcão/WhatsApp";
            const notes = document.getElementById('cart-notes').value.trim();
            const total = calculateTotal();
            const horaAtual = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
            const orderId = Math.floor(100 + Math.random() * 900);

            // Objeto do Pedido para o Painel da Cozinha
            const novoPedidoObj = {
                id: orderId,
                cliente: clientName,
                hora: horaAtual,
                itens: cart.map(i => `${i.qty}x ${i.name}`),
                obs: notes,
                total: total,
                pagamento: "Pix / Cartão",
                status: "novos"
            };

            // 1. Adiciona localmente e envia via BroadcastChannel para outras abas
            adicionarPedidoAoGestor(novoPedidoObj, true, true);

            // 2. Monta texto formatado para o WhatsApp
            let message = `Ol%C3%A1!%20Gostaria%20de%20fazer%20o%20seguinte%20pedido%20(Ped%20%23${orderId})%3A%0A%0A`;
            message += `*Cliente:* ${encodeURIComponent(clientName)}%0A%0A`;
            
            cart.forEach(item => {
                const subtotal = item.price * item.qty;
                message += `%E2%96%AA%20${item.qty}x%20${encodeURIComponent(item.name)}%20(R%24%20${subtotal.toFixed(2).replace('.', ',')})%0A`;
            });

            if (notes) {
                message += `%0A%F0%9F%93%9D%20*Obs:*%20${encodeURIComponent(notes)}%0A`;
            }

            message += `%0A%F0%9F%92%B2%20*Total%20da%20Comanda:*%20R%24%20${total.toFixed(2).replace('.', ',')}`;

            // Limpa comanda
            cart = [];
            document.getElementById('cart-client-name').value = '';
            document.getElementById('cart-notes').value = '';
            updateCartUI();
            toggleCartModal(false);

            showToast("Pedido enviado com sucesso!", 'success');
            window.open(`https://wa.me/5527999999999?text=${message}`, '_blank');
        }

        function adicionarPedidoAoGestor(pedido, notificar = false, broadcast = false) {
            if (gestorPedidos.some(p => p.id === pedido.id)) return;

            gestorPedidos.push(pedido);
            renderizarCardKanban(pedido);
            salvarPedidosLocalStorage();

            if (notificar) tocarAlertaSom();

            if (broadcast) {
                orderChannel.postMessage({ type: 'NOVO_PEDIDO', pedido });
            }
        }

        function renderizarCardKanban(pedido) {
            const card = document.createElement('div');
            card.id = `kanban-pedido-${pedido.id}`;
            card.className = "bg-brand-card p-4 rounded-xl border border-brand-border shadow-md transition-all hover:border-brand-gold/40 relative group";

            card.innerHTML = `
                <div class="flex justify-between items-start mb-2">
                    <div>
                        <span class="font-bold text-white text-sm">#${pedido.id}</span>
                        <p class="text-xs font-semibold text-brand-gold">${pedido.cliente}</p>
                    </div>
                    <div class="flex items-center gap-1">
                        <button onclick="imprimirComanda(${pedido.id})" title="Imprimir Comanda" class="text-slate-400 hover:text-brand-gold p-1 text-xs transition">
                            <i class="fa-solid fa-print"></i>
                        </button>
                        <button onclick="removerPedidoGestor(${pedido.id})" title="Cancelar Pedido" class="text-slate-500 hover:text-red-400 p-1 text-xs transition">
                            <i class="fa-solid fa-xmark"></i>
                        </button>
                        <span class="text-[10px] text-slate-400 bg-slate-900 px-2 py-0.5 rounded-md border border-slate-800 ml-1">${pedido.hora}</span>
                    </div>
                </div>
                
                <div class="text-xs text-slate-300 border-t border-b border-slate-800/80 py-2 my-2 space-y-1">
                    ${pedido.itens.map(item => `<div>• ${item}</div>`).join('')}
                </div>

                ${pedido.obs ? `<div class="text-xs text-brand-red font-medium mb-2 bg-brand-red/10 p-1.5 rounded border border-brand-red/20"><strong>Obs:</strong> ${pedido.obs}</div>` : ''}

                <div class="flex justify-between items-center text-xs font-bold text-slate-200 mb-3">
                    <span>R$ ${pedido.total.toFixed(2).replace('.', ',')}</span>
                    <span class="text-[10px] text-slate-400">${pedido.pagamento}</span>
                </div>

                <div id="acoes-${pedido.id}" class="flex gap-2"></div>
            `;

            moverCardColuna(card, pedido.status, pedido.id);
        }

        function moverCardColuna(cardElement, status, id) {
            const acoes = cardElement.querySelector(`#acoes-${id}`);
            if (!acoes) return;

            acoes.innerHTML = '';

            const pedido = gestorPedidos.find(p => p.id === id);
            if (pedido) pedido.status = status;

            if (status === 'novos') {
                document.getElementById('col-novos').prepend(cardElement);
                acoes.innerHTML = `
                    <button onclick="alterarStatusPedido(${id}, 'preparo')" class="w-full bg-blue-600 hover:bg-blue-700 text-white font-bold py-1.5 rounded-lg text-xs transition">
                        Aceitar & Preparar →
                    </button>
                `;
            } else if (status === 'preparo') {
                document.getElementById('col-preparo').prepend(cardElement);
                acoes.innerHTML = `
                    <button onclick="alterarStatusPedido(${id}, 'pronto')" class="w-full bg-emerald-600 hover:bg-emerald-700 text-white font-bold py-1.5 rounded-lg text-xs transition">
                        Marcar como Pronto →
                    </button>
                `;
            } else if (status === 'pronto') {
                document.getElementById('col-pronto').prepend(cardElement);
                acoes.innerHTML = `
                    <button onclick="alterarStatusPedido(${id}, 'concluido')" class="w-full bg-slate-800 hover:bg-slate-700 text-slate-200 font-bold py-1.5 rounded-lg text-xs border border-brand-border transition">
                        Finalizar Pedido ✓
                    </button>
                `;
            } else if (status === 'concluido') {
                document.getElementById('col-concluido').prepend(cardElement);
                acoes.innerHTML = `
                    <span class="text-xs text-emerald-400 font-bold flex items-center justify-center w-full py-1">
                        <i class="fa-solid fa-circle-check mr-1"></i> Entregue
                    </span>
                `;
            }

            atualizarContadoresKanban();
            salvarPedidosLocalStorage();
        }

        function alterarStatusPedido(id, novoStatus, broadcast = true) {
            const card = document.getElementById(`kanban-pedido-${id}`);
            if (card) {
                moverCardColuna(card, novoStatus, id);
                if (broadcast) {
                    orderChannel.postMessage({ type: 'STATUS_ALTERADO', id, novoStatus });
                }
            }
        }

        function removerPedidoGestor(id, broadcast = true) {
            gestorPedidos = gestorPedidos.filter(p => p.id !== id);
            const card = document.getElementById(`kanban-pedido-${id}`);
            if (card) card.remove();

            atualizarContadoresKanban();
            salvarPedidosLocalStorage();

            if (broadcast) {
                orderChannel.postMessage({ type: 'PEDIDO_REMOVIDO', id });
            }
            showToast(`Pedido #${id} removido.`, 'info');
        }

        function limparConcluidos(broadcast = true) {
            const concluidos = gestorPedidos.filter(p => p.status === 'concluido');
            if (concluidos.length === 0) {
                showToast("Nenhum pedido concluído para limpar.", "info");
                return;
            }

            gestorPedidos = gestorPedidos.filter(p => p.status !== 'concluido');
            document.getElementById('col-concluido').innerHTML = '';

            atualizarContadoresKanban();
            salvarPedidosLocalStorage();

            if (broadcast) {
                orderChannel.postMessage({ type: 'CONCLUIDOS_LIMPOS' });
            }
            showToast("Coluna de concluídos limpa!", "success");
        }

        function atualizarContadoresKanban() {
            document.getElementById('count-novos').innerText = document.getElementById('col-novos').children.length;

        function salvarPedidosLocalStorage() {
            localStorage.setItem('julius_kanban_pedidos', JSON.stringify(gestorPedidos));
        }

        function carregarPedidosLocalStorage() {
            const salvos = localStorage.getItem('julius_kanban_pedidos');
            if (salvos) {
                try {
                    const lista = JSON.parse(salvos);
                    gestorPedidos = [];
                    lista.forEach(p => adicionarPedidoAoGestor(p, false, false));
                } catch(e) { console.error(e); }
            }
        }

        function filterCategory(category) {
            currentCategory = category;
            document.querySelectorAll('.filter-btn').forEach(btn => btn.classList.remove('active'));
            const btnAtivo = document.getElementById('filter-' + category);
            if (btnAtivo) btnAtivo.classList.add('active');
            applyMenuFilters();
        }

        function filterMenuSearch() {
            applyMenuFilters();
        }

        function applyMenuFilters() {
            const query = (document.getElementById('search-input')?.value || '').toLowerCase().trim();
            const items = document.querySelectorAll('.menu-item');

            items.forEach(item => {
                const matchesCategory = (currentCategory === 'todos' || item.classList.contains('category-' + currentCategory));
                const matchesSearch = !query || item.innerText.toLowerCase().includes(query);

                if (matchesCategory && matchesSearch) {
                    item.classList.remove('hidden');
                } else {
                    item.classList.add('hidden');
                }
            });
        }
